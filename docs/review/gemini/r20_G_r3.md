def make_bvh_patch(bvh, x_center, z_center, rx, rz, y_standoff, n_ang=24, squircle=0.12):

    evaluated `head` BVHTree at `y_standoff` metres proud of the skin surface.

            ca = math.cos(ang)

            zv = z_center + sa * rz * rf * sq

    return bm

            u = -1.0 + 2.0 * iu / (nu - 1)
            abs_u = abs(u)

            # Clean blunt cut with a gentle slight-V taper at the outer corners (c5)
            corner_lift = 0.035 * H * (abs_u ** 2.2)
            z_end = z_bot + corner_lift
            z = z_top * (1.0 - v) + z_end * v

            if v < 0.25:
                t = v / 0.25
                half_w = H * (0.44 + 0.045 * math.sin(t * math.pi * 0.5))
            else:
                t = (v - 0.25) / 0.75
                half_w = H * (0.485 + 0.020 * math.sin(t * math.pi * 0.6) - 0.025 * (t ** 1.6))

            x = u * half_w

            if v < 0.28:
                wrap_depth = H * (0.35 * (1.0 - 0.82 * (abs_u ** 1.6)))
                y_base = 0.10 * H + wrap_depth
            else:
                t = (v - 0.28) / 0.72
                curve_back = H * (0.28 + 0.050 * math.sin(t * math.pi * 0.8))
                edge_cup = H * 0.14 * (abs_u ** 1.7)
                # Subtle vertical strand flutes in Y only (fading to 0 at bottom edge so hem is crisp!)
                sheet_wave = 0.008 * H * math.cos(u * math.pi * 4.0) * math.sin(v * math.pi)
                y_base = curve_back - edge_cup + sheet_wave

            row.append(bm.verts.new((x, y_base, z)))
        loops.append(row)

    loft_loops(bm, loops, closed_loop=False)
    back_obj = new_mesh_obj("back_hair", bm, mat=mats["hair"], shade_smooth=True)
    add_solidify(back_obj, thickness=0.044 * H, offset=-1.0)
    add_subsurf(back_obj, levels=2)

    # Inner dark-brown shadow curtain (`back_hair_inner` in `hair_shadow` #6E3C3A)
    bm_in = bmesh.new()
    in_loops = []
    for iv in range(10):
        v = iv / 9.0
        z = (z_top - 0.08 * H) * (1.0 - v) + (z_bot + 0.04 * H) * v
        half_w = H * (0.44 - 0.03 * v)
        row = []
        for iu in range(9):
            u = -1.0 + 2.0 * iu / 8.0
            x = u * half_w
            y = H * (0.16 + 0.06 * v - 0.12 * (abs(u) ** 1.5))
            row.append(bm_in.verts.new((x, y, z)))
        in_loops.append(row)
    loft_loops(bm_in, in_loops, closed_loop=False)
    in_obj = new_mesh_obj("back_hair_inner", bm_in, mat=mats["hair_shadow"], shade_smooth=True)
    add_solidify(in_obj, thickness=0.025 * H, offset=-1.0)
    add_subsurf(in_obj, levels=1)

    return back_obj


def build_ponytail(P, mats):
    """
    Rebuilt for R3 to match c3, c4, c5, c8, c9:
    - Emerges high at the back of the crown under the bow knot (`z_root = z_chin + 0.73 H`)
      and drapes down in a graceful, fuller vertical S-curve parallel to the back hair (c4),
      ending in a clean V-point at `P.z_ponytail_tip` (c5).
    - Keeps its upper width slightly narrower than the bow tail spacing so both teal `bow_tail_L/R`
      ribbons read clearly from the knot all the way down in back view (`c5`, `c9`)!
    """
    H = P.H
    bm = bmesh.new()

    z_root = P.z_chin + 0.73 * H
    z_tip = P.z_ponytail_tip
    y_root = 0.44 * H

    n_rings = 18
    n_seg = 24
    loops = []

    for ir in range(n_rings):
        t = ir / (n_rings - 1)

        z = z_root * (1.0 - t) + z_tip * t + 0.038 * H * math.sin(t * math.pi * 0.55) * (1.0 - t)
        # Graceful S-curve that stays fuller and closer to the back hair in the lower half (c4)
        s_arch = math.sin(t * math.pi * 0.75) * 0.155 * H
        yc = y_root + s_arch

        if t < 0.25:
            s = t / 0.25
            rad = H * (0.090 + 0.125 * math.sin(s * math.pi * 0.5))
        elif t < 0.68:
            s = (t - 0.25) / 0.43
            rad = H * (0.215 - 0.025 * (s ** 1.2))
        else:
            s = (t - 0.68) / 0.32
            rad = H * 0.190 * ((1.0 - s) ** 1.15)
        rad = max(0.005 * H, rad)

        tilt = 0.22 * (1.0 - 1.8 * t)
        rot = Euler((tilt, 0.0, 0.0), 'XYZ').to_matrix()

        ring = []
        for i in range(n_seg):
            theta = 2.0 * math.pi * i / n_seg
            clump_flute = 1.0 + 0.09 * math.cos(5.0 * theta + 0.4 * t) * math.sin(t * math.pi)
            rx = rad * 0.94 * clump_flute
            ry = rad * 0.88 * clump_flute
            local = Vector((rx * math.cos(theta), ry * math.sin(theta), 0.0))
            pt = Vector((0.0, yc, z)) + (rot @ local)
            ring.append(bm.verts.new(pt))
        loops.append(ring)

    loft_loops(bm, loops, cap_start=True, cap_end=True, closed_loop=True)
    pt_obj = new_mesh_obj("ponytail", bm, mat=mats["hair"], shade_smooth=True)
    add_subsurf(pt_obj, levels=1)
    return pt_obj
